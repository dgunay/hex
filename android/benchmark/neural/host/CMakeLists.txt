cmake_minimum_required(VERSION 3.22)
project(hextyping_host CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE ON CACHE BOOL "" FORCE)
# llama.cpp is cloned into the work folder (android/.dev/lm/llama.cpp), not committed.
set(LLAMA_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../../.dev/lm/llama.cpp CACHE PATH "llama.cpp checkout")
add_subdirectory(${LLAMA_DIR} llama.cpp)
find_package(JNI REQUIRED)
set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/../../../app/src/main/cpp/hextyping.cpp)
add_library(hextyping SHARED ${SRC})
target_include_directories(hextyping PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(hextyping PRIVATE llama)
add_executable(hextyping-cli ${SRC})
target_compile_definitions(hextyping-cli PRIVATE HEX_TYPING_CLI)
target_include_directories(hextyping-cli PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(hextyping-cli PRIVATE llama)
